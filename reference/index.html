    @model_validator(mode="after")
    def _check_consistency(self) -> "ClusteringConfig":
        """Valida a faixa de k e a coerência entre listas de variáveis."""
        if self.k_min > self.k_max:
            raise ValueError("k_min não pode ser maior que k_max")
        unknown = set(self.log_features) - set(self.features)
        if unknown:
            raise ValueError(f"log_features fora de features: {sorted(unknown)}")
        return self

    class IntegrityConfig(_FrozenModel):
    """Limiares dos testes de integridade da base municipal."""

    min_coverage: dict[str, float]
    max_null_share: float = Field(default=0.02, ge=0, le=1)
    null_share_overrides: dict[str, float] = Field(default_factory=dict)

    @model_validator(mode="after")
    def _check_fractions(self) -> "IntegrityConfig":
        """Garante que todos os limiares estejam entre 0 e 1."""
        values = [*self.min_coverage.values(), *self.null_share_overrides.values()]
        if any(not 0 <= value <= 1 for value in values):
            raise ValueError("limiares de cobertura e nulos devem estar entre 0 e 1")
        return self

    def load_settings(config_dir: Path = CONFIG_DIR) -> Settings:
    """Carrega todos os YAML de configuração e os valida com Pydantic.

    Parameters
    ----------
    config_dir : Path, optional
        Diretório com os arquivos YAML, by default ``configs/``.

    Returns
    -------
    Settings
        Configuração validada e imutável.

    Raises
    ------
    ConfigurationError
        Se um arquivo estiver ausente, malformado, repetir uma seção ou
        violar o esquema.

    Examples
    --------
    >>> settings = load_settings()
    >>> settings.project.random_seed
    42
    """
    merged: dict[str, Any] = {}
    for filename in CONFIG_FILES:
        section = _read_yaml(config_dir / filename)
        repeated = merged.keys() & section.keys()
        if repeated:
            raise ConfigurationError(f"Seções repetidas em {filename}: {sorted(repeated)}")
        merged.update(section)
    try:
        return Settings(**merged)
    except ValidationError as exc:
        raise ConfigurationError(f"Configuração inválida:\n{exc}") from exc

    def download_file(
    url: str,
    destination: Path,
    *,
    timeout: int = 120,
    force: bool = False,
    chunk_size: int = 1 << 20,
    retries: int = 5,
    backoff: float = 2.0,
) -> ManifestEntry:
    """Baixa ``url`` para ``destination`` apenas se ainda não estiver em cache.

    O cache é válido quando o arquivo existe e seu SHA-256 confere com o
    manifesto. O download é gravado em um ``.part`` e movido de forma atômica,
    de modo que uma falha no meio nunca deixa um arquivo truncado como se
    fosse válido.

    Parameters
    ----------
    url : str
        Endereço de origem.
    destination : Path
        Arquivo de destino na camada bronze.
    timeout : int, optional
        Timeout das requisições em segundos, by default 120.
    force : bool, optional
        Ignora o cache e baixa novamente, by default False.
    chunk_size : int, optional
        Tamanho do bloco de escrita em bytes, by default 1 MiB.
    retries : int, optional
        Tentativas extras após falhas transitórias (conexão resetada, timeout,
        HTTP 429/5xx); cada nova tentativa retoma do ``.part`` via ``Range``,
        by default 5.
    backoff : float, optional
        Base da espera exponencial entre tentativas, em segundos, by default 2.0.

    Returns
    -------
    ManifestEntry
        Registro de proveniência do arquivo.

    Raises
    ------
    IngestionError
        Se a requisição falhar (rede, status HTTP de erro, timeout).

    Examples
    --------
    >>> download_file(url, Path("data/bronze/fonte/2023/arquivo.zip"))  # doctest: +SKIP
    """
    configure_system_certificates()
    destination.parent.mkdir(parents=True, exist_ok=True)

    cached = _read_valid_cache(destination, url, force)
    if cached is not None:
        return cached

    temporary = destination.with_name(destination.name + ".part")
    logger.info("Baixando %s", url)
    if force:
        temporary.unlink(missing_ok=True)
    _download_with_retries(url, temporary, timeout, chunk_size, retries, backoff)

    temporary.replace(destination)
    return register_file(destination, url)

    def aggregate_schools_by_municipality(schools: pl.LazyFrame | pl.DataFrame) -> pl.DataFrame:
    """Agrega escolas em atividade por município.

    As proporções (``*_share``) são médias de indicadores 0/1 entre as escolas
    em atividade; valores nulos do indicador são ignorados na média.

    Parameters
    ----------
    schools : pl.LazyFrame or pl.DataFrame
        Escolas com as colunas de ``CENSO_RAW_COLUMNS`` já numéricas.

    Returns
    -------
    pl.DataFrame
        Uma linha por município, ordenada por ``code_muni``.

    Examples
    --------
    >>> aggregate_schools_by_municipality(schools).columns[:2]  # doctest: +SKIP
    ['code_muni', 'n_schools']
    """
    active = schools.lazy().filter(pl.col("TP_SITUACAO_FUNCIONAMENTO") == ACTIVE_SCHOOL)
    aggregated = active.group_by("CO_MUNICIPIO").agg(
        pl.len().cast(pl.Int64).alias("n_schools"),
        pl.col("QT_MAT_BAS").sum().cast(pl.Int64).alias("enrollments"),
        pl.col("QT_DOC_BAS").sum().cast(pl.Int64).alias("teachers"),
        (pl.col("TP_LOCALIZACAO") == RURAL_LOCATION).mean().alias("rural_share"),
        (pl.col("TP_DEPENDENCIA") == MUNICIPAL_NETWORK).mean().alias("municipal_network_share"),
        *[
            pl.col(raw).cast(pl.Float64).mean().alias(name)
            for raw, name in CENSO_FLAG_COLUMNS.items()
        ],
    )
    return (
        aggregated.with_columns(
            pl.when(pl.col("teachers") > 0)
            .then(pl.col("enrollments") / pl.col("teachers"))
            .alias("students_per_teacher"),
            (pl.col("enrollments") / pl.col("n_schools")).alias("enrollments_per_school"),
        )
        .rename({"CO_MUNICIPIO": KEY})
        .sort(KEY)
        .collect()
    )

    def extract_censo_csv(zip_path: Path, staging_dir: Path) -> Path:
    """Extrai o CSV da Educação Básica do ``.zip`` para a área de staging.

    A extração é idempotente: se o arquivo já existe com o mesmo tamanho, é
    reaproveitado. O bronze (o ``.zip``) nunca é modificado.

    Parameters
    ----------
    zip_path : Path
        Arquivo ``.zip`` dos microdados na camada bronze.
    staging_dir : Path
        Diretório temporário (descartável) para o CSV extraído.

    Returns
    -------
    Path
        Caminho do CSV extraído.

    Raises
    ------
    DataValidationError
        Se o ``.zip`` não contiver o CSV ``microdados_ed_basica_AAAA.csv``.
    """
    staging_dir.mkdir(parents=True, exist_ok=True)
    with zipfile.ZipFile(zip_path) as archive:
        members = [name for name in archive.namelist() if _CSV_PATTERN.search(name)]
        if not members:
            raise DataValidationError(
                f"CSV 'microdados_ed_basica_AAAA.csv' não encontrado em {zip_path.name}"
            )
        member = members[0]
        target = staging_dir / Path(member).name  # só o nome: evita path traversal
        if target.exists() and target.stat().st_size == archive.getinfo(member).file_size:
            logger.info("CSV já extraído, reaproveitando: %s", target.name)
            return target
        logger.info("Extraindo %s", member)
        with archive.open(member) as source, target.open("wb") as sink:
            shutil.copyfileobj(source, sink)
    return target

    report.duplicate_keys["gold"] = count_duplicate_keys(gold)

        if share > limit:
            report.violations.append(
                f"{share:.1%} de nulos em '{column}' acima do limite de {limit:.1%}"
            )

    return report

    def assess_stability(
    matrix: np.ndarray,
    reference_labels: np.ndarray,
    k: int,
    repeats: int,
    fraction: float,
    n_init: int,
    seed: int,

    """Estima a estabilidade dos clusters por subamostragem.

    Em cada repetição ajusta-se o KMeans em uma subamostra aleatória, atribuem-se
    todos os municípios ao modelo obtido e compara-se com a partição de
    referência pelo índice de Rand ajustado (ARI). Valores próximos de 1 indicam

    Parameters
    ----------
    matrix : np.ndarray
        Matriz padronizada.
    reference_labels : np.ndarray
        Partição do ajuste completo.
    k : int
        Número de clusters.
    repeats : int
        Número de subamostragens.
    fraction : float
        Fração do total em cada subamostra (0 a 1, exclusivo).
    n_init : int
        Reinicializações do KMeans.
    seed : int
        Semente.

    Returns
    -------
    StabilitySummary
        Média do ARI e intervalo de 95% (percentis 2,5 e 97,5).
    """
    rng = np.random.default_rng(seed)
    size = max(k, int(fraction * len(matrix)))
    scores = np.empty(repeats)
    for repeat in range(repeats):
        index = rng.choice(len(matrix), size=size, replace=False)
        model = KMeans(n_clusters=k, n_init=n_init, random_state=seed + repeat).fit(matrix[index])  # pyright: ignore[reportArgumentType]

    return StabilitySummary(
        mean=float(scores.mean()), ci_low=float(low), ci_high=float(high), n_repeats=repeats

    def bootstrap_median_ci(
    values: np.ndarray, n_resamples: int, seed: int, level: float = 0.95
) -> tuple[float, float]:
    """Intervalo de confiança percentílico (bootstrap) para a mediana.

    Parameters
    ----------
    values : np.ndarray
        Amostra sem nulos.
    n_resamples : int
        Número de reamostragens com reposição.
    seed : int
        Semente.
    level : float, optional
        Nível de confiança, by default 0.95.

    Returns
    -------
    tuple[float, float]
        Limites inferior e superior; ``(nan, nan)`` para amostra vazia.
    """
    if len(values) == 0:
        return float("nan"), float("nan")
    rng = np.random.default_rng(seed)
    samples = rng.choice(values, size=(n_resamples, len(values)), replace=True)
    medians = np.median(samples, axis=1)
    alpha = (1 - level) / 2
    low, high = np.quantile(medians, [alpha, 1 - alpha])
    return float(low), float(high)

    """Resume cada cluster: tamanho, mediana das variáveis e do desfecho com IC 95%.

    Parameters
    ----------
    df : pl.DataFrame
        Tabela com a coluna ``cluster`` (municípios clusterizados).
    features : list[str]
        Variáveis da clusterização (escala original).
    outcomes : list[str]
        Desfechos não usados na clusterização (ex.: IDEB).
    n_resamples : int
        Reamostragens do bootstrap.
    seed : int
        Semente.

    Returns
    -------
    pl.DataFrame
        Uma linha por cluster, com ``n``, ``share``, ``<variável>_median`` e,
        para cada desfecho, ``_median``, ``_ci_low``, ``_ci_high`` e ``_n``.
    """
    total = df.height
    rows = []
    for (cluster,), group in df.group_by(CLUSTER, maintain_order=True):
        row: dict[str, object] = {
            CLUSTER: cluster,
            "n": group.height,
            "share": group.height / total,
        }
        for name in features:
            row[f"{name}_median"] = group[name].median()
        for name in outcomes:
            valid = group[name].drop_nulls().to_numpy()
            low, high = bootstrap_median_ci(valid, n_resamples, seed)
            row[f"{name}_median"] = float(np.median(valid)) if len(valid) else None
            row[f"{name}_ci_low"] = low
            row[f"{name}_ci_high"] = high
            row[f"{name}_n"] = len(valid)
        rows.append(row)
    return pl.DataFrame(rows, infer_schema_length=None).sort(CLUSTER)

    ----------
    values : np.ndarray

        Matriz de vizinhança (padronizada por linha).

    -------
    pl.DataFrame
        Colunas ``local_i``, ``quadrant``, ``p_value``, ``p_adjusted`` e
        ``lisa_class`` (rótulo em pt-BR), na ordem de ``values``.
    """
    local = Moran_Local(values, weights, permutations=permutations, seed=seed)
    p_values = np.asarray(local.p_sim, dtype=np.float64)
    p_adjusted = false_discovery_control(p_values, method="bh") if apply_fdr else p_values
    quadrants = np.asarray(local.q, dtype=np.int64)
    classes = [
        LISA_QUADRANT_LABELS[int(q)] if p < alpha else LISA_NOT_SIGNIFICANT
        for q, p in zip(quadrants, p_adjusted, strict=True)
    ]
    return pl.DataFrame(

            "quadrant": quadrants,
            "p_value": p_values,
            "p_adjusted": np.asarray(p_adjusted, dtype=np.float64),
            "lisa_class": classes,
        }
    )

    def compute_cluster_cohesion(labels: np.ndarray, weights: W) -> pl.DataFrame:
    """Compara a fração observada de vizinhos no mesmo cluster com a esperada ao acaso.

    Para cada município, mede-se a fração de vizinhos Queen que pertencem ao seu
    próprio cluster. Sob distribuição espacial aleatória, essa fração esperada é
    a participação do cluster no total. Razão > 1 indica que o cluster forma
    manchas contíguas (tipologia com dimensão territorial).

    Parameters
    ----------
    labels : np.ndarray
        Rótulo de cluster de cada município, alinhado às linhas de ``weights``.
    weights : W
        Matriz de vizinhança sem ilhas.

    Returns
    -------
    pl.DataFrame
        Colunas ``cluster``, ``observed_share``, ``expected_share`` e ``ratio``.
    """
    adjacency = (weights.sparse > 0).astype(np.float64)  # pyright: ignore[reportAttributeAccessIssue]
    clusters = np.unique(labels)
    indicator = (labels[:, None] == clusters[None, :]).astype(np.float64)
    neighbors_in_cluster = adjacency @ indicator  # (n, k): vizinhos em cada cluster
    own = neighbors_in_cluster[np.arange(len(labels)), np.searchsorted(clusters, labels)]
    degree = np.asarray(adjacency.sum(axis=1)).ravel()
    same_share = own / degree

    rows = []
    for cluster in clusters:
        members = labels == cluster
        expected = float(members.mean())
        observed = float(same_share[members].mean())
        rows.append(
            {
                "cluster": int(cluster),
                "observed_share": observed,
                "expected_share": expected,
                "ratio": observed / expected if expected > 0 else float("nan"),
            }
        )
    return pl.DataFrame(rows)
